# Step 4: Configure PostgreSQL for Point-in-Time Recovery (PITR)

# First, create directories for WAL archives and backups
mkdir C:\PostgreSQL\wal_archives
mkdir C:\PostgreSQL\backups
mkdir C:\PostgreSQL\logs

# Edit postgresql.conf file
# Location: C:\Program Files\PostgreSQL\15\data\postgresql.conf

# Backup the original configuration
copy "C:\Program Files\PostgreSQL\15\data\postgresql.conf" "C:\Program Files\PostgreSQL\15\data\postgresql.conf.backup"

# Add these settings to postgresql.conf:

# WRITE AHEAD LOG
wal_level = replica
archive_mode = on
archive_command = 'copy "%p" "C:\\PostgreSQL\\wal_archives\\%f"'
archive_timeout = 300

# CHECKPOINTS
checkpoint_completion_target = 0.9
wal_buffers = 16MB
checkpoint_timeout = 15min
max_wal_size = 4GB
min_wal_size = 2GB

# LOGGING
log_destination = 'stderr'
logging_collector = on
log_directory = 'C:/PostgreSQL/logs'
log_filename = 'postgresql-%Y-%m-%d_%H%M%S.log'
log_rotation_age = 1d
log_rotation_size = 100MB
log_min_duration_statement = 1000
log_connections = on
log_disconnections = on
log_duration = on
log_statement = 'mod'

# PERFORMANCE
shared_buffers = 256MB
effective_cache_size = 1GB
work_mem = 4MB
maintenance_work_mem = 64MB

# REPLICATION
max_wal_senders = 3
wal_keep_size = 2GB
